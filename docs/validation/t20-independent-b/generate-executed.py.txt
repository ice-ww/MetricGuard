import json,traceback
from pathlib import Path
from metricguard.evaluation import EvaluationRunner,file_hash
from metricguard.semantic_rubric import digest
from experiments.t18.freeze import build_case,configuration
root=Path.cwd(); plan_path=root/'artifacts/t20-new-instance-plan.json'; plan=json.loads(plan_path.read_text(encoding='utf-8'))
assert digest(plan['recipes'])==plan['recipe_hash']
output=root/'artifacts/t20-independent-b'; output.mkdir(exist_ok=False); data=output/'data'
config=configuration(plan['image']); assert config['code_ref']=='726a1071382b215ba26661c15b1a7111c1338921'
owner=EvaluationRunner(root/'artifacts/evaluation-registry',repository=root,snapshot_root=data)
generation={'status':'running','plan_sha256':file_hash(plan_path),'recipe_hash':plan['recipe_hash'],'family':plan['family'],'version':plan['version'],'configuration':config,'configuration_hash':digest(config),'product_runs':0,'diagnosis_runs':0,'provenance':[],'failures':[]}
def save(): (output/'generation.json').write_text(json.dumps(generation,ensure_ascii=False,indent=2),encoding='utf-8')
save(); cases=[]
try:
 for index,recipe in enumerate(plan['recipes']):
  owner.verify_configuration(config)
  case,provenance=build_case(recipe,data)
  cases.append(case); generation['provenance'].append(provenance); save()
  print(json.dumps({'generated':index+1,'reference_agreement':True}),flush=True)
 owner.verify_configuration(config)
 assert len({c['snapshot_id'] for c in cases})==24
 frozen=owner.freeze(plan['version'],plan['family'],cases,config)
 generation.update(status='frozen',freeze=frozen,locked_status=owner.status(plan['version']),unchanged_cases_hash=digest(cases),reference_agreements=24,unique_snapshot_ids=24)
 (data/'provenance.json').write_text(json.dumps(generation['provenance'],ensure_ascii=False,indent=2),encoding='utf-8')
 (output/'public-tasks.json').write_text(json.dumps([c['public'] for c in cases],ensure_ascii=False,indent=2),encoding='utf-8')
 save()
 print(json.dumps({'status':generation['status'],'freeze':frozen,'configuration_hash':digest(config),'cases_hash':digest(cases),'reference_agreements':24,'product_runs':0,'generation_file':str(output/'generation.json')}),flush=True)
except BaseException as e:
 generation['status']='generation_failed'; generation['failures'].append({'type':type(e).__name__,'message':str(e),'traceback':traceback.format_exc(),'completed_cases':len(cases)})
 save(); raise
